//======================================  WHEN  ==========================================================	
	@When("I send a {string} request to {string}")
	public void the_user_sends_a_httpMethod_request(String httpMethod, String authEndpoint) {
		switch (httpMethod.toUpperCase()) {

		case "GET":

			// ---------------------------------------------
			// FIRST SEND THE REQUEST
			// ---------------------------------------------

			String token = context.getToken();

			Response response = authMeService.getCurrentUserAuthMe(AuthEndpoints.AuthMe);

			context.setResponse(response);

			System.out.println("HTTP Method : " + httpMethod);
			System.out.println("Status Code : " + response.statusCode());
			System.out.println("Response Body : ");
			System.out.println(response.asPrettyString());

			// ---------------------------------------------
			// THEN CHECK THE RESPONSE STATUS
			// ---------------------------------------------

			if (context.getResponse().getStatusCode() == HttpConstants.OK) {

				authResponse = authMeService.authMeUser(response);
				System.out.println("User success : " + authResponse.isSuccess());
				System.out.println("User ID : " + authResponse.getUser().getUserId());
				System.out.println("User Email : " + authResponse.getUser().getEmail());
				System.out.println("User iat : " + authResponse.getUser().getIat());
				System.out.println("User exp : " + authResponse.getUser().getExp());

				// Store Auth Me User ID
				Integer authMeUserId = authResponse.getUser().getUserId();
				context.setAuthMeUserId(authMeUserId);

				// Store Auth Me Email
				String authMeEmail = authResponse.getUser().getEmail();
				context.setAuthMeEmail(authMeEmail);
				// Store response and token in the shared context
				// context.setResponse(response);

				// -----------------------------------------
				// SAVE TOKEN ONLY AFTER SUCCESSFUL RESPONSE
				// -----------------------------------------
				// Store it securely in your ThreadLocal TokenManager

				if (token != null && !token.isBlank()) {

					TokenManager.setToken(token);
					System.out.println("Token saved in TokenManager");
				}

				System.out.println("========================================");
				System.out.println();

			} else {

				baseErrorResponse = authMeService.authMeUserUsingPojoFailure(response);
				System.out.println("Error : " + baseErrorResponse.getError());
			}

			System.out.println("========================================");
			System.out.println();

			break;

		default:

			throw new IllegalArgumentException("Unsupported HTTP method: " + httpMethod);
		}

	}

	@Then("store the token")
	public void save_token() {
		// Retrieve the token stored by LoginSteps
		// Automatically fetches the token for the current thread with 'Bearer ' prefix
		// String bearerHeader = TokenManager.getBearerToken();;

		String token = context.getResponse().jsonPath().getString("token");
		// context.setToken(token);

		// Store it securely in your ThreadLocal TokenManager

		if (token != null && !token.isBlank()) {

			TokenManager.setToken(token);

			System.out.println("Token saved in TokenManager : " + token);
		}
		token = TokenManager.getToken();
		Assert.assertNotNull(token, "Token is null in TokenManager");
		Assert.assertFalse(token.isBlank(), "Token is empty in TokenManager");

		System.out.println("Token retrieved from TokenManager");

		System.out.println("========================================");
		System.out.println();

	}

	@Then("the {string} value should be less than the {string} value")
	public void value_should_be_less_than_another_value(String firstFieldPath, String secondFieldPath) {
		// Extract the values from the response body using JSON path
		Object firstValue = context.getResponse().jsonPath().get(firstFieldPath);
		Object secondValue = context.getResponse().jsonPath().get(secondFieldPath);

		// 1. Verify both fields actually exist in the response
		Assert.assertNotNull(firstValue, "Field '" + firstFieldPath + "' is missing or null");
		Assert.assertNotNull(secondValue, "Field '" + secondFieldPath + "' is missing or null");

		// 2. Verify both fields contain numeric values
		Assert.assertTrue(firstValue instanceof Number, "Field '" + firstFieldPath + "' is not a number");
		Assert.assertTrue(secondValue instanceof Number, "Field '" + secondFieldPath + "' is not a number");

		// 3. Cast to Number and compare as long values to prevent any Integer/Long
		// mismatch errors
		long firstNum = ((Number) firstValue).longValue();
		long secondNum = ((Number) secondValue).longValue();

		// 4. Perform the assertion check
		Assert.assertTrue(firstNum < secondNum, "Assertion Failed: '" + firstFieldPath + "' (" + firstNum
				+ ") is NOT less than '" + secondFieldPath + "' (" + secondNum + ")");
	}
